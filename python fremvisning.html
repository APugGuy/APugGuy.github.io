En enkel oppgaveplanlegger

Denne konsollappen lar brukeren legge til, vise og slette skoleoppgaver ved hjelp av et lite meny-system.

# Liste for å lagre oppgaver
oppgaver = []

def legg_til():
		fag = input("Fag: ")
		frist = input("Frist: ")
		oppgaver.append({"fag": fag, "frist": frist})
		print("Oppgave lagt til!\n")

def vis():
		print("\n--- OPPGAVER ---")
		for o in oppgaver:
				print(f"{o['fag']} - Frist: {o['frist']}")
		print()

def slett():
		if not oppgaver:
				print("Ingen oppgaver å slette.\n")
				return

		print("\n--- SLETT OPPGAVE ---")
		for i, o in enumerate(oppgaver):
				print(f"[{i}] {o['fag']} - Frist: {o['frist']}")
		print()

		try:
				idx = int(input("Hvilken oppgave (nummer) skal slettes?: "))
				if 0 <= idx < len(oppgaver):
						slettet = oppgaver.pop(idx)
						print(f"Slettet: {slettet['fag']} - {slettet['frist']}\n")
				else:
						print("Ugyldig nummer.\n")
		except ValueError:
				print("Skriv et tall.\n")

def main():
		while True:
				print("1: Legg til oppgave")
				print("2: Vis oppgaver")
				print("3: Slett oppgave")
				print("4: Avslutt")

				valg = input("Velg: ")

				if valg == "1":
						legg_til()
				elif valg == "2":
						vis()
				elif valg == "3":
						slett()
				elif valg == "4":
						print("Hadet!")
						break
				else:
						print("Ugyldig valg.\n")

main()
